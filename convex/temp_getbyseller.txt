export const getBySeller = query({
    args: { sellerId: v.string() },
    handler: async (ctx, args) => {
        // Find all listings for the seller
        const listings = await ctx.db
            .query("listings")
            .filter(q => q.eq(q.field("sellerId"), args.sellerId))
            .collect();

        const listingIds = listings.map(l => l._id.toString());
        if (listingIds.length === 0) return [];

        // In Convex we can't easily query by array membership efficiently without an index on a specific field,
        // but we can query all reviews and filter, or do multiple queries. Let's do multiple queries since
        // this is an MVP or filter all reviews. Doing multiple queries is safer for small sets.
        let allReviews: any[] = [];
        for (const listingId of listingIds) {
            const reviews = await ctx.db
                .query("reviews")
                .withIndex("by_listing", q => q.eq("listingId", listingId))
                .collect();
            allReviews = allReviews.concat(reviews);
        }

        return allReviews.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
    }
});
